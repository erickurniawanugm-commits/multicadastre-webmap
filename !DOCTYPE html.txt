// 1. SETUP PETA
const map = new maplibregl.Map({
    container: "map",
    center: [110.367, -7.795],
    zoom: 16,
    pitch: 55,
    bearing: -10,
    style: {
        version: 8,
        sources: { basemap: { type: "raster", tiles: ["https://tile.openstreetmap.org/{z}/{x}/{y}.png"], tileSize: 256 } },
        layers: [{ id: "basemap-layer", type: "raster", source: "basemap" }]
    }
});
map.addControl(new maplibregl.NavigationControl(), 'top-right');

// 2. SETUP DRAW TOOLS
const draw = new MapboxDraw({
    displayControlsDefault: false,
    controls: { point: false, line_string: false, polygon: false, trash: false },
    styles: [
        { 'id': 'gl-draw-line', 'type': 'line', 'filter': ['all', ['==', '$type', 'LineString'], ['!=', 'mode', 'static']], 'paint': { 'line-color': '#e74c3c', 'line-width': 4 } },
        { 'id': 'gl-draw-polygon-fill', 'type': 'fill', 'filter': ['all', ['==', '$type', 'Polygon'], ['!=', 'mode', 'static']], 'paint': { 'fill-color': '#3bb2d0', 'fill-opacity': 0.3 } },
        { 'id': 'gl-draw-polygon-stroke', 'type': 'line', 'filter': ['all', ['==', '$type', 'Polygon'], ['!=', 'mode', 'static']], 'paint': { 'line-color': '#3bb2d0', 'line-width': 3 } },
        { 'id': 'gl-draw-point', 'type': 'circle', 'filter': ['all', ['==', '$type', 'Point'], ['!=', 'mode', 'static']], 'paint': { 'circle-radius': 7, 'circle-color': '#fbb03b' } }
    ]
});
map.addControl(draw);

function startDraw(mode) { draw.changeMode(mode); }
function deleteSelected() { draw.trash(); updateDrawInfo(); }
map.on('draw.create', updateDrawInfo);
map.on('draw.delete', updateDrawInfo);
map.on('draw.update', updateDrawInfo);

function updateDrawInfo(e) {
    const data = draw.getAll();
    const panel = document.getElementById('measure-result');
    if (data.features.length > 0) {
        let html = '<table class="info-table">';
        data.features.forEach((feature, index) => {
            let info = "";
            let tipe = feature.geometry.type;
            if (tipe === 'Polygon') info = `${Math.round(turf.area(feature))} m²`;
            else if (tipe === 'LineString') info = `${(turf.length(feature, {units: 'kilometers'}) * 1000).toFixed(2)} m`;
            else if (tipe === 'Point') info = `Lat: ${feature.geometry.coordinates[1].toFixed(5)}`;
            html += `<tr><td>#${index+1} (${tipe})</td><td>${info}</td></tr>`;
        });
        html += '</table>';
        panel.innerHTML = html;
    } else {
        panel.innerHTML = '<div class="info-placeholder">Hasil ukur muncul di sini.</div>';
    }
}
function downloadDrawings() {
    const data = draw.getAll();
    if (data.features.length === 0) return alert("Belum ada gambar!");
    const a = document.createElement("a");
    a.href = URL.createObjectURL(new Blob([JSON.stringify(data)], {type: "application/json"}));
    a.download = "hasil_gambar_sipamer.geojson";
    a.click();
}

// 3. LOGIKA OGC API FEATURES (NEW!)
function loadOGCLayer(url) {
    // Hapus layer lama jika ada
    if (map.getSource('ogc-source')) {
        map.removeLayer('ogc-layer');
        map.removeSource('ogc-source');
    }

    // Tampilkan Loading
    const panel = document.getElementById('ogc-panel');
    const oriText = panel.innerHTML;
    // panel.innerHTML += "<small>Loading data...</small>";

    fetch(url)
        .then(r => r.json())
        .then(data => {
            // Cek apakah hasil valid GeoJSON
            if(!data.features) { alert("Respon API bukan GeoJSON valid!"); return; }

            map.addSource('ogc-source', { type: 'geojson', data: data });
            
            // Tambahkan Layer (Titik/Garis/Poligon OGC)
            map.addLayer({
                'id': 'ogc-layer',
                'type': 'circle', // Asumsi data titik dulu (paling umum)
                'source': 'ogc-source',
                'paint': {
                    'circle-radius': 6,
                    'circle-color': '#0000ff', // Biru
                    'circle-stroke-width': 2,
                    'circle-stroke-color': '#ffffff'
                }
            });

            // Zoom ke Data OGC
            const bbox = turf.bbox(data);
            map.fitBounds(bbox, { padding: 50 });
            alert(`Berhasil memuat ${data.features.length} fitur dari OGC API!`);
        })
        .catch(e => alert("Gagal mengambil data dari URL API: " + e));
}

function loadUserOGC() {
    const url = document.getElementById('ogcUrl').value.trim();
    if(!url) return alert("Masukkan URL dulu!");
    loadOGCLayer(url);
}

function loadDemoOGC() {
    // URL Demo Publik (Daraa, Syria - Transportasi)
    const demoUrl = "https://demo.ldproxy.net/daraa/collections/TransportationGroundCrv/items?f=json&limit=50";
    loadOGCLayer(demoUrl);
}


// 4. LOGIKA UTAMA (DATA LOCAL)
let bidangData = null;
let hoveredStateId = null; 
let clickedStateId = null; 
let hoveredBuildingId = null; 
let clickedBuildingId = null; 
let KamusGedung = {}; 
let currentPopup = null; 

function setBasemap(type, btnElement) {
    const url = (type === "osm") ? "https://tile.openstreetmap.org/{z}/{x}/{y}.png" : "https://server.arcgisonline.com/ArcGIS/rest/services/World_Imagery/MapServer/tile/{z}/{y}/{x}";
    map.getSource("basemap").setTiles([url]);
    document.querySelectorAll('.btn-map').forEach(b => b.classList.remove('active'));
    if(btnElement) btnElement.classList.add('active');
}

map.on("load", () => {
    fetch("cityobject_data.json").then(r => r.json()).then(data => {
        const items = data.features ? data.features : data;
        items.forEach(item => { const p = item.properties ? item.properties : item; if(p.id) KamusGedung[String(p.id)] = p.name || "Tanpa Nama"; });
    }).catch(e => console.log("CityObject missing"));

    fetch("BidangTanah.geojson").then(r => r.json()).then(data => {
        bidangData = data;
        map.addSource("bidang", { type: "geojson", data: data, promoteId: "NIB" });
        map.addLayer({ id: "bidang-fill", type: "fill", source: "bidang", paint: { "fill-color": "#ff0000", "fill-opacity": 0.2 } });
        map.addLayer({ id: "bidang-outline", type: "line", source: "bidang", paint: { "line-color": "#333", "line-width": 1 } });
        map.addLayer({
            id: "bidang-highlight", type: "line", source: "bidang",
            paint: {
                "line-color": ["case", ["boolean", ["feature-state", "selected"], false], "#00ffff", ["boolean", ["feature-state", "hover"], false], "#ffff00", "transparent"],
                "line-width": ["case", ["boolean", ["feature-state", "selected"], false], 4, ["boolean", ["feature-state", "hover"], false], 3, 0]
            }
        });
        if (data.features.length > 0) map.fitBounds(turf.bbox(data), { padding: 50 });
    });

    fetch("kotabaru2.geojson").then(r => r.json()).then(dataBangunan => {
        dataBangunan.features.forEach((feature, index) => { feature.id = index; }); 
        map.addSource("bangunan", { type: "geojson", data: dataBangunan });
        map.addLayer({
            id: "bangunan-3d", type: "fill-extrusion", source: "bangunan",
            paint: {
                "fill-extrusion-color": ["case", ["boolean", ["feature-state", "selected"], false], "#e74c3c", ["boolean", ["feature-state", "hover"], false], "#ffff00", "#aaaaaa"],
                "fill-extrusion-height": ["get", "tinggi_z"], "fill-extrusion-base": 0, "fill-extrusion-opacity": 0.95
            }
        });
    });
});

map.on("click", (e) => {
    if (draw.getMode() !== 'simple_select') return;
    if (e.originalEvent.target.closest('.btn-tool')) return; // Abaikan klik tombol draw

    const featuresBangunan = map.queryRenderedFeatures(e.point, { layers: ['bangunan-3d'] });
    const featuresTanah = map.queryRenderedFeatures(e.point, { layers: ['bidang-fill'] });

    // KLIK BANGUNAN
    if (featuresBangunan.length > 0) {
        const feature = featuresBangunan[0];
        if (clickedBuildingId !== null) map.setFeatureState({ source: "bangunan", id: clickedBuildingId }, { selected: false });
        clickedBuildingId = feature.id;
        map.setFeatureState({ source: "bangunan", id: clickedBuildingId }, { selected: true });

        if (currentPopup) currentPopup.remove();
        const props = feature.properties;
        let linkId = String(props.id_cityobject || props.cityobject_id); 
        let nama = (linkId && KamusGedung[linkId]) ? KamusGedung[linkId] : (props.gmlid || "Tanpa Nama");

        const html = `<div class="popup-header">${nama}</div><div class="popup-body"><div class="popup-row"><span class="popup-label">Tinggi:</span><span>${props.tinggi_z}m</span></div><div class="popup-row"><span class="popup-label">ID:</span><span>${props.gmlid}</span></div></div>`;
        currentPopup = new maplibregl.Popup().setLngLat(e.lngLat).setHTML(html).addTo(map);
        return;
    }

    // KLIK TANAH
    if (featuresTanah.length > 0) {
        const feature = featuresTanah[0];
        if (clickedStateId !== null) map.setFeatureState({ source: "bidang", id: clickedStateId }, { selected: false });
        clickedStateId = feature.id;
        map.setFeatureState({ source: "bidang", id: clickedStateId }, { selected: true });

        const props = feature.properties;
        document.getElementById('info-panel').innerHTML = `
            <table class="info-table">
                <tr><td>NIB</td><td>${props.NIB}</td></tr>
                <tr><td>Jenis Hak</td><td>${props.TIPEHAK || '-'}</td></tr>
                <tr><td>Luas</td><td>${props.LUAS} m²</td></tr>
            </table>`;
        return; 
    }

    // KLIK KOSONG
    if (clickedBuildingId !== null) { map.setFeatureState({ source: "bangunan", id: clickedBuildingId }, { selected: false }); clickedBuildingId = null; }
    if (clickedStateId !== null) { map.setFeatureState({ source: "bidang", id: clickedStateId }, { selected: false }); clickedStateId = null; }
    if (currentPopup) { currentPopup.remove(); currentPopup = null; }
    document.getElementById('info-panel').innerHTML = '<div class="info-placeholder">Klik objek untuk detail.</div>';
});

// HOVER EFFECT
map.on("mousemove", "bidang-fill", (e) => {
    if (draw.getMode() !== 'simple_select') return;
    map.getCanvas().style.cursor = "pointer";
    if (hoveredStateId !== null) map.setFeatureState({ source: "bidang", id: hoveredStateId }, { hover: false });
    hoveredStateId = e.features[0].id;
    map.setFeatureState({ source: "bidang", id: hoveredStateId }, { hover: true });
});
map.on("mouseleave", "bidang-fill", () => {
    map.getCanvas().style.cursor = "";
    if (hoveredStateId !== null) map.setFeatureState({ source: "bidang", id: hoveredStateId }, { hover: false });
    hoveredStateId = null;
});
map.on("mousemove", "bangunan-3d", (e) => {
    if (draw.getMode() !== 'simple_select') return;
    map.getCanvas().style.cursor = "pointer";
    if (hoveredBuildingId !== null) map.setFeatureState({ source: "bangunan", id: hoveredBuildingId }, { hover: false });
    hoveredBuildingId = e.features[0].id;
    map.setFeatureState({ source: "bangunan", id: hoveredBuildingId }, { hover: true });
});
map.on("mouseleave", "bangunan-3d", () => {
    map.getCanvas().style.cursor = "";
    if (hoveredBuildingId !== null) map.setFeatureState({ source: "bangunan", id: hoveredBuildingId }, { hover: false });
    hoveredBuildingId = null;
});

// SEARCH
function searchNIB() {
    const input = document.getElementById("searchBox").value.trim();
    if (!input || !bidangData) return;
    const found = bidangData.features.find(f => String(f.properties.NIB) === input);
    if (!found) return alert("NIB tidak ditemukan");
    map.fitBounds(turf.bbox(found), { padding: 100 });
    document.getElementById('info-panel').innerHTML = `<table class="info-table"><tr><td>NIB</td><td>${found.properties.NIB}</td></tr><tr><td>Tipe Hak</td><td>${found.properties.TIPEHAK||'-'}</td></tr><tr><td>Luas</td><td>${found.properties.LUAS} m²</td></tr></table>`;
    if (clickedStateId !== null) map.setFeatureState({ source: "bidang", id: clickedStateId }, { selected: false });
    clickedStateId = found.properties.NIB;
    map.setFeatureState({ source: "bidang", id: clickedStateId }, { selected: true });
}
document.getElementById("searchBox").addEventListener("keypress", (e) => { if(e.key==="Enter") searchNIB() });
